<script>


var yesButton = document.getElementById("yes");

var noButton = document.getElementById("no");

var hint = document.getElementById("hint");


var questionCard = document.getElementById("questionCard");

var successCard = document.getElementById("successCard");


var noClicks = 0;


var messages = [

  "Are you sure? 🥺",

  "Really sure? 🙈",

  "Think about the dinner... 🍝",

  "I'll be very sad... 🥹",

  "One tiny dinner? ❤️",

  "Pretty please? 🥺💕",

  "The No button is getting shy now...",

  "Okay... you know what to do 😌❤️"

];



noButton.addEventListener("click", function() {


  noClicks++;


  var noScale = Math.max(0.12, 1 - noClicks * 0.13);

  var yesScale = Math.min(2.4, 1 + noClicks * 0.16);


  noButton.style.transform = "scale(" + noScale + ")";

  yesButton.style.transform = "scale(" + yesScale + ")";


  noButton.style.opacity = Math.max(

    0.25,

    1 - noClicks * 0.1

  );


  var messageIndex = Math.min(

    noClicks - 1,

    messages.length - 1

  );


  hint.textContent = messages[messageIndex];


});



yesButton.addEventListener("click", function() {


  questionCard.style.display = "none";

  successCard.style.display = "block";


  createHearts(25);


});



function createHeart() {


  var heart = document.createElement("div");


  heart.className = "heart";


  if (Math.random() > 0.5) {

    heart.textContent = "♥️";

  } else {

    heart.textContent = "♡";

  }


  heart.style.left = Math.random() * 100 + "vw";


  heart.style.fontSize =

    (14 + Math.random() * 24) + "px";


  heart.style.animationDuration =

    (5 + Math.random() * 6) + "s";


  document.body.appendChild(heart);


  setTimeout(function() {

    heart.remove();

  }, 12000);


}



function createHearts(amount) {


  for (var i = 0; i < amount; i++) {


    setTimeout(function() {

      createHeart();

    }, i * 120);


  }


}



setInterval(function() {

  createHeart();

}, 900);
